root ::= object ws
object ::= "{" ws schema-member ws "," ws speech-member ws "," ws expression-member ws "," ws gesture-member ws "," ws urgency-member ws "}" | "{" ws schema-member ws "," ws speech-member ws "," ws gaze-member ws "," ws expression-member ws "," ws gesture-member ws "," ws urgency-member ws "}"
schema-member ::= "\"schema_version\"" ws ":" ws "1"
speech-member ::= "\"speech\"" ws ":" ws string
gaze-member ::= "\"gaze_target\"" ws ":" ws ("null" | gaze-object)
gaze-object ::= "{" ws "\"kind\"" ws ":" ws "\"tracked_entity\"" ws "," ws "\"entity_id\"" ws ":" ws entity-id ws "}"
expression-member ::= "\"expression\"" ws ":" ws expression
gesture-member ::= "\"gesture\"" ws ":" ws gesture
urgency-member ::= "\"urgency\"" ws ":" ws urgency
expression ::= "\"neutral\"" | "\"attentive\"" | "\"curious\"" | "\"pleased\"" | "\"concerned\"" | "\"surprised\""
gesture ::= "\"none\"" | "\"nod\"" | "\"small_head_tilt\"" | "\"recoil\""
urgency ::= "\"low\"" | "\"normal\"" | "\"high\""
entity-id ::= "\"entity-" digit+ "\""
string ::= "\"" char* "\""
char ::= [^"\\\x7F\x00-\x1F] | "\\" (["\\/bfnrt] | "u" hex hex hex hex)
hex ::= [0-9a-fA-F]
digit ::= [0-9]
ws ::= [ \t\n\r]*
